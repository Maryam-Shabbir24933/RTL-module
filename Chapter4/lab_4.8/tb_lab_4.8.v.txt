
`timescale 1ns/1ps
module tb_fifo;
    reg  [7:0] d_in;
    reg        d_in_valid, d_out_req, clk;
    wire [7:0] d_out;
    wire       full, empty;

    fifo dut (.d_in(d_in), .d_in_valid(d_in_valid), .d_out(d_out),
              .d_out_req(d_out_req), .clk(clk), .full(full), .empty(empty));

    initial clk = 1'b0;
    always #5 clk = ~clk;

    reg [7:0] mq [0:127];
    reg [7:0] last_out;
    integer mw, mr, mcount, errors, k, seed;
    reg do_wr, do_rd;

    task cycle(input v, input [7:0] d, input r);
        begin
            do_wr = v && (mcount < 128);
            do_rd = r && (mcount > 0);
            d_in = d; d_in_valid = v; d_out_req = r;
            @(posedge clk); #1;

            if (do_rd) begin
                if (d_out !== mq[mr % 128]) begin
                    $display("ERROR: read got 0x%02h expected 0x%02h (t=%0t)", d_out, mq[mr % 128], $time);
                    errors = errors + 1;
                end
                last_out = d_out;
                mr = mr + 1;
            end else if (d_out !== last_out) begin
                $display("ERROR: d_out changed without a read (t=%0t)", $time);
                errors = errors + 1;
            end

            if (do_wr) begin
                mq[mw % 128] = d;
                mw = mw + 1;
            end
            mcount = mw - mr;

            if (full !== (mcount == 128)) begin
                $display("ERROR: full=%b with count=%0d (t=%0t)", full, mcount, $time);
                errors = errors + 1;
            end
            if (empty !== (mcount == 0)) begin
                $display("ERROR: empty=%b with count=%0d (t=%0t)", empty, mcount, $time);
                errors = errors + 1;
            end
        end
    endtask

    initial begin
        $dumpfile("fifo.vcd");
        $dumpvars(0, tb_fifo);
        errors = 0; mw = 0; mr = 0; mcount = 0; seed = 11;
        last_out = 8'hxx;
        d_in = 0; d_in_valid = 0; d_out_req = 0;
        @(negedge clk);

        cycle(0, 8'h00, 0);
        if (!empty || full) begin
            $display("ERROR: bad flags after reset");
            errors = errors + 1;
        end

        // fill all 128 entries back-to-back
        for (k = 0; k < 128; k = k + 1) cycle(1, 3 * k + 1, 0);

        // writes while full must be ignored
        for (k = 0; k < 3; k = k + 1) cycle(1, 8'hEE, 0);

        // drain all 128 back-to-back
        for (k = 0; k < 128; k = k + 1) cycle(0, 8'h00, 1);

        // reads while empty must be ignored
        for (k = 0; k < 3; k = k + 1) cycle(0, 8'h00, 1);

        // simultaneous read and write
        cycle(1, 8'hA1, 0);
        for (k = 0; k < 20; k = k + 1) cycle(1, 8'hB0 + k, 1);
        for (k = 0; k < 2;  k = k + 1) cycle(0, 8'h00, 1);

        // write-heavy then read-heavy random traffic
        for (k = 0; k < 600; k = k + 1)
            cycle(($random(seed) & 3) != 0, $random(seed), ($random(seed) & 3) == 0);
        for (k = 0; k < 600; k = k + 1)
            cycle(($random(seed) & 3) == 0, $random(seed), ($random(seed) & 3) != 0);
        for (k = 0; k < 200; k = k + 1)
            cycle($random(seed), $random(seed), $random(seed));

        if (errors == 0) $display("*** TEST PASSED ***");
        else             $display("*** TEST FAILED: %0d error(s) ***", errors);
        $finish;
    end

    initial begin
        #1000000;
        $display("*** TIMEOUT ***");
        $finish;
    end
endmodule
