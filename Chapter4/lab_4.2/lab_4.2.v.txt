module counter(rst, clk, up, down, count);
  input wire clk;
  input wire rst;
  input wire up;
  input wire down;
  output reg [7:0] count;
  
  always @(posedge clk or negedge rst) begin
    if(!rst) begin
      count <= 8'b0;
    end
      else begin
        if(up) 
          count <= count +1;
          else if(down)
            count <= count - 1;
          else 
            count <= count;
        end
  end
endmodule
