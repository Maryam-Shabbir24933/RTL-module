module majority_voter(clk, in, out);

  input wire clk;
  input wire in;
  output reg out;

  reg sample1;
  reg sample2;
  reg sample3;

  always @(posedge clk) begin
      sample1 <= in;
      sample2 <= sample1;
      sample3 <= sample2;

      if ((sample1 + sample2 + sample3) >= 2)
        out <= 1;
      else
        out <= 0;

    end


endmodule
